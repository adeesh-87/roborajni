# Lean build for the LSP-vs-Graphify experiment: 64-bit, C11, no static analysis. Upstream build: see git history.
cmake_minimum_required(VERSION 3.16)
project(canard_exp C CXX)
enable_testing()
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_EXTENSIONS OFF)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_CXX_STANDARD 20)
add_library(unity STATIC lib/unity/src/unity.c)
target_include_directories(unity SYSTEM PUBLIC lib/unity/src)
target_compile_definitions(unity PUBLIC UNITY_INCLUDE_DOUBLE=1 UNITY_SUPPORT_64=1 UNITY_SHORTHAND_AS_RAW=1)
include_directories(SYSTEM lib/cavl2)
file(GLOB INTRUSIVE tests/src/test_intrusive_*.c tests/src/test_helpers.c tests/src/test_exp_*.c)
foreach(src ${INTRUSIVE})
  get_filename_component(n ${src} NAME_WE)
  add_executable(${n} ${src})
  target_include_directories(${n} PUBLIC libcanard tests/src)
  target_link_libraries(${n} unity)
  add_test(NAME ${n} COMMAND ${n})
endforeach()
file(GLOB API tests/src/test_api_*.cpp)
foreach(src ${API})
  get_filename_component(n ${src} NAME_WE)
  add_executable(${n} libcanard/canard.c ${src})
  target_include_directories(${n} PUBLIC libcanard tests/src)
  target_link_libraries(${n} unity)
  add_test(NAME ${n} COMMAND ${n})
endforeach()
